// =========================================================
// HELPERS
// =========================================================
function h1(text) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_1,
    spacing: { before: 360, after: 200 },
    border: { bottom: { style: BorderStyle.SINGLE, size: 8, color: NAVY, space: 4 } },
    children: [new TextRun({ text, font: "Arial", size: 32, bold: true, color: NAVY })]
  });
}

function h2(text, color = BLUE) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_2,
    spacing: { before: 280, after: 160 },
    children: [new TextRun({ text, font: "Arial", size: 26, bold: true, color })]
  });
}

function h3(text, color = NAVY) {
  return new Paragraph({
    heading: HeadingLevel.HEADING_3,
    spacing: { before: 200, after: 100 },
    children: [new TextRun({ text, font: "Arial", size: 22, bold: true, color })]
  });
}

function para(runs, spacing = { before: 60, after: 60 }, align = AlignmentType.LEFT) {
  return new Paragraph({ spacing, alignment: align, children: runs });
}

function txt(text, opts = {}) {
  return new TextRun({ text, font: "Arial", size: opts.size || 22, bold: opts.bold || false, color: opts.color || "000000", italics: opts.italic || false });
}

function pageBreak() {
  return new Paragraph({ children: [new PageBreak()] });
}

function spacer(before = 80, after = 80) {
  return new Paragraph({ spacing: { before, after }, children: [new TextRun("")] });
}

function divider(color = NAVY) {
  return new Paragraph({
    spacing: { before: 120, after: 120 },
    border: { bottom: { style: BorderStyle.SINGLE, size: 6, color, space: 1 } },
    children: [new TextRun("")]
  });
}

function numberedQuestion(num, chinese, pinyin, meaning) {
  const children = [
    txt(`Câu ${num}: `, { bold: true, color: NAVY }),
    txt(chinese, { bold: true, color: RED, size: 24 }),
  ];
  if (pinyin) children.push(txt(`  (${pinyin})`, { italic: true, color: DARKGRAY }));
  if (meaning) children.push(txt(`  → ${meaning}`, { color: GREEN }));
  return new Paragraph({ spacing: { before: 80, after: 40 }, children });
}

function answerLine(label, value, extra = "") {
  return new Paragraph({
    spacing: { before: 30, after: 30 },
    indent: { left: 360 },
    children: [
      txt(`${label} `, { bold: true, color: ORANGE }),
      txt(value, { bold: true, color: RED }),
      ...(extra ? [txt(`   ${extra}`, { color: DARKGRAY, italic: true })] : [])
    ]
  });
}

function note(text) {
  return new Paragraph({
    spacing: { before: 30, after: 40 },
    indent: { left: 360 },
    children: [
      txt("📝 ", { size: 20 }),
      txt(text, { color: DARKGRAY, size: 20, italic: true })
    ]
  });
}

// Multiple choice question builder
function mcQuestion(num, questionText, options, correctLetter, explanation) {
  const items = [];
  // Question
  items.push(new Paragraph({
    spacing: { before: 100, after: 40 },
    children: [
      txt(`${num}. `, { bold: true, color: NAVY, size: 23 }),
      txt(questionText, { size: 23 })
    ]
  }));
  // Options
  const optLabels = ["A", "B", "C", "D"];
  options.forEach((opt, i) => {
    const isCorrect = optLabels[i] === correctLetter;
    items.push(new Paragraph({
      spacing: { before: 20, after: 20 },
      indent: { left: 360 },
      children: [
        txt(`${optLabels[i]}. `, { bold: isCorrect, color: isCorrect ? GREEN : DARKGRAY, size: 22 }),
        txt(opt, { bold: isCorrect, color: isCorrect ? GREEN : "000000", size: 22 })
      ]
    }));
  });
  // Answer + explanation
  items.push(new Paragraph({
    spacing: { before: 30, after: 20 },
    indent: { left: 360 },
    children: [
      txt("✔ Đáp án: ", { bold: true, color: GREEN, size: 21 }),
      txt(correctLetter, { bold: true, color: RED, size: 21 }),
      txt("   |   ", { size: 21, color: DARKGRAY }),
      txt(explanation, { italic: true, color: DARKGRAY, size: 20 })
    ]
  }));
  return items;
}

// Grammar rule box using table
function ruleBox(title, content, bgColor = LIGHTBLUE, borderColor = BLUE) {
  const border = { style: BorderStyle.SINGLE, size: 3, color: borderColor };
  const borders = { top: border, bottom: border, left: border, right: border };
  return new Table({
    width: { size: 9360, type: WidthType.DXA },
    columnWidths: [9360],
    margins: { top: 80, bottom: 80 },
    rows: [
      new TableRow({
        children: [new TableCell({
          borders,
          width: { size: 9360, type: WidthType.DXA },
          shading: { fill: borderColor, type: ShadingType.CLEAR },
          margins: { top: 80, bottom: 80, left: 200, right: 200 },
          children: [new Paragraph({ children: [txt(title, { bold: true, color: WHITE, size: 22 })] })]
        })]
      }),
      new TableRow({
        children: [new TableCell({
          borders,
          width: { size: 9360, type: WidthType.DXA },
          shading: { fill: bgColor, type: ShadingType.CLEAR },
          margins: { top: 100, bottom: 100, left: 200, right: 200 },
          children: content.map(c => new Paragraph({ spacing: { before: 40, after: 40 }, children: [txt(c, { size: 21 })] }))
        })]
      })
    ]
  });
}

// 2-col comparison table
function twoColTable(headers, rows, col1w = 3000, col2w = 6360) {
  const hBorder = { style: BorderStyle.SINGLE, size: 2, color: "AAAAAA" };
  const bAll = { top: hBorder, bottom: hBorder, left: hBorder, right: hBorder };
  const tableRows = [];
  // Header
  tableRows.push(new TableRow({
    children: headers.map((h, i) => new TableCell({
      borders: bAll,
      width: { size: i === 0 ? col1w : col2w, type: WidthType.DXA },
      shading: { fill: NAVY, type: ShadingType.CLEAR },
      margins: { top: 80, bottom: 80, left: 120, right: 120 },
      children: [new Paragraph({ children: [txt(h, { bold: true, color: WHITE, size: 22 })] })]
    }))
  }));
  // Rows
  rows.forEach((row, ri) => {
    tableRows.push(new TableRow({
      children: row.map((cell, ci) => new TableCell({
        borders: bAll,
        width: { size: ci === 0 ? col1w : col2w, type: WidthType.DXA },
        shading: { fill: ri % 2 === 0 ? WHITE : GRAY, type: ShadingType.CLEAR },
        margins: { top: 60, bottom: 60, left: 120, right: 120 },
        children: [new Paragraph({ children: [txt(cell, { size: 21 })] })]
      }))
    }));
  });
  return new Table({ width: { size: 9360, type: WidthType.DXA }, columnWidths: [col1w, col2w], rows: tableRows });
}

// 4-col table
function fourColTable(headers, rows, widths = [2000, 2500, 2500, 2360]) {
  const hBorder = { style: BorderStyle.SINGLE, size: 2, color: "AAAAAA" };
  const bAll = { top: hBorder, bottom: hBorder, left: hBorder, right: hBorder };
  const tableRows = [];
  tableRows.push(new TableRow({
    children: headers.map((h, i) => new TableCell({
      borders: bAll,
      width: { size: widths[i], type: WidthType.DXA },
      shading: { fill: NAVY, type: ShadingType.CLEAR },
      margins: { top: 80, bottom: 80, left: 120, right: 120 },
      children: [new Paragraph({ alignment: AlignmentType.CENTER, children: [txt(h, { bold: true, color: WHITE, size: 20 })] })]
    }))
  }));
  rows.forEach((row, ri) => {
    tableRows.push(new TableRow({
      children: row.map((cell, ci) => new TableCell({
        borders: bAll,
        width: { size: widths[ci], type: WidthType.DXA },
        shading: { fill: ri % 2 === 0 ? WHITE : GRAY, type: ShadingType.CLEAR },
        margins: { top: 60, bottom: 60, left: 120, right: 120 },
        children: [new Paragraph({ children: [txt(cell, { size: 20, bold: ci === 0 })] })]
      }))
    }));
  });
  return new Table({ width: { size: 9360, type: WidthType.DXA }, columnWidths: widths, rows: tableRows });
}
