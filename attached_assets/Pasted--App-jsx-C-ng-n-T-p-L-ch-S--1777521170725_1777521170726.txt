setCurrentQ(0); setSelected(null);

if (selected !== null) return;

if (idx === questions[currentQ].ans) setScore(s => s + 1);
setAnswers(prev => […prev, { correct: idx === questions[currentQ].ans }]);

if (currentQ + 1 >= questions.length) setFinished(true);
else { setCurrentQ(q => q + 1); setSelected(null); setShowExplain(false); }
